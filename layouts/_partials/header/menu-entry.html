{{- /* Renders one top-level header menu entry. Context: dict "item" MenuEntry "variant" ("desktop"|"hamburger") "mode" string Dropdown when the item has native .Children (menu entries with parent = this id). isHamburger — inside the mobile drawer (forces click; hamburger-menu-item). isMore — desktop identifier "more" (menu-dropdown-more under-bar panel). */ -}} {{- $item := .item -}} {{- $variant := .variant | default "desktop" -}} {{- $id := $item.Identifier -}} {{- $isHamburger := eq $variant "hamburger" -}} {{- $isMore := and (eq $id "more") (not $isHamburger) -}} {{- $mode := .mode | default "hover" -}} {{- if $isHamburger -}} {{- $mode = "click" -}} {{- end -}} {{- if not $item.HasChildren -}} {{- if $isHamburger -}}
{{- partial "header/menu-icon.html" $id -}} {{ $item.Name }}
{{- else -}} {{- partial "header/menu-icon.html" $id -}} {{ $item.Name }} {{- end -}} {{- else -}} {{- $icon := partial "header/menu-icon.html" $id -}} {{- $label := printf `%s%s` $icon ($item.URL | relURL) $item.Name -}} {{- $lis := "" -}} {{- range $item.Children -}} {{- $childIcon := partial "header/menu-icon.html" .Identifier -}} {{- $lis = printf `%s
  • %s%s
    • ` $lis $childIcon (.URL | relURL) .Name -}} {{- end -}} {{- $panel := printf `
      %s
    ` $lis | safeHTML -}} {{- $wrapperClass := cond $isHamburger "hamburger-menu-item menu-dropdown" (cond $isMore "menu-item menu-dropdown menu-dropdown-more" "menu-item menu-dropdown") -}} {{- partial "dropdown.html" (dict "id" (printf "%s-dd-%s" $variant $id) "label" $label "panel" $panel "mode" $mode "class" $wrapperClass "ariaLabel" (printf "Toggle %s submenu" $item.Name) ) -}} {{- end -}}
