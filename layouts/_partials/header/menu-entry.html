{{- /* One menu entry. A plain item renders as an inline link; when the item has native .Children it becomes a dropdown whose panel lists those children. Context: dict "item" MenuEntry "variant" ("desktop"|"hamburger") "mode" string. */ -}} {{- $item := .item -}} {{- $variant := .variant | default "desktop" -}} {{- $id := $item.Identifier -}} {{- $isMore := and (eq $id "more") (eq $variant "desktop") -}} {{- $mode := .mode | default "hover" -}} {{- $iconPath := $item.Params.icon.path | default "" -}} {{- $iconClass := $item.Params.icon.class | default (printf "menu-icon menu-%s-icon" $id) -}} {{- $iconID := $item.Params.icon.id -}} {{- $iconAttrs := $item.Params.icon.attrs -}} {{- $icon := partial "svg-icon.html" (dict "path" $iconPath "class" $iconClass "id" $iconID "attrs" $iconAttrs) -}} {{- if not $item.HasChildren -}} {{- if eq $variant "hamburger" -}}
{{ $icon }} {{ $item.Name }}
{{- else -}} {{ $icon }} {{ $item.Name }} {{- end -}} {{- else -}} {{- $label := printf `%s%s` $icon ($item.URL | relURL) $item.Name -}} {{- $lis := "" -}} {{- range $item.Children -}} {{- $childIcon := partial "svg-icon.html" (dict "path" (.Params.icon.path | default "") "class" (.Params.icon.class | default (printf "menu-icon menu-%s-icon" .Identifier)) "id" .Params.icon.id "attrs" .Params.icon.attrs) -}} {{- $lis = printf `%s
  • %s%s
    • ` $lis $childIcon (.URL | relURL) .Name -}} {{- end -}} {{- $panel := printf `
      %s
    ` $lis | safeHTML -}} {{- $wrapperClass := cond $isMore "menu-item menu-dropdown menu-dropdown-more" (cond (eq $variant "hamburger") "hamburger-menu-item menu-dropdown" "menu-item menu-dropdown") -}} {{- partial "dropdown.html" (dict "id" (printf "%s-dd-%s" $variant $id) "label" $label "panel" $panel "mode" $mode "class" $wrapperClass "ariaLabel" (printf "Toggle %s submenu" $item.Name) ) -}} {{- end -}}
