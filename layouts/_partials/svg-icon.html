{{- /* Resolve an icon reference. Config no longer carries a sprite `id`; the id is derived at build time and published in `/sprite_map.json` (see scripts/build-icons.js). This partial reads that map ONCE and caches it globally in hugo.Store, then turns a configured svg `path` into a sprite `#` reference. Priority: 1. `path` found in the sprite map → "#" (uses the bundled sprite). 2. `path` that ends in ".svg" → the raw file itself (standalone icon / live preview). 3. "svg/placeholder.svg" (universal default). Every returned value goes through relURL, so it resolves from any page depth (e.g. /posts/). Parameter (flat dict, e.g. a menu entry's `.Params.icon`): path (string): the icon's key, e.g. "ui-oval/home-03.svg". This is both the sprite-map lookup key and, failing that, the direct file URL. Return: string $iconURL — for an inline ``, root-relative. */ -}} {{- $sprite := (site.Params.icons.sprite | default "svg/sprite/svg-icon-sprite.svg") -}} {{- $map := hugo.Store.Get "alice-sprite-map" -}} {{- if not $map -}} {{- $mapPath := "static/svg/sprite/sprite_map.json" -}} {{- $map = $mapPath | os.ReadFile | transform.Unmarshal -}} {{- hugo.Store.Set "alice-sprite-map" $map -}} {{- end -}} {{- $iconPath := .path -}} {{- $iconURL := "svg/placeholder.svg" | relURL -}} {{- $id := index $map $iconPath -}} {{- if $id -}} {{- $iconURL = printf "%s#%s" ($sprite | relURL) $id -}} {{- else -}} {{- if strings.HasSuffix $iconPath ".svg" -}} {{- $iconURL = $iconPath | relURL -}} {{- end -}} {{- end -}} {{- return $iconURL -}}
