{{- /* Math, TikZ, and algorithm rendering, in one entry point. baseof.html calls this partial when a page (or the site) sets params.tex. Hugo only treats the exact string "mathml" as pure MathML and silently renders anything else as HTML, so reject values outside the three supported spellings before they get that far. */ -}} {{- $hugoOutput := ($tex.hugo | default dict).output | default "mathml" -}} {{- if not (in (slice "mathml" "html" "htmlAndMathml") $hugoOutput) -}} {{- errorf "Alice: params.tex.hugo.output must be \"mathml\", \"html\", or \"htmlAndMathml\"; got %q." $hugoOutput -}} {{- end -}} {{- $isHugo := eq $engine "hugo" -}} {{- /* HTML output needs the stylesheet. An algorithm needs it too: pseudocode renders its math through the client-side katex, whose classes live in that same stylesheet. */ -}} {{- $hugoNeedsKatexCSS := and $isHugo (or (ne $hugoOutput "mathml") $algorithm) -}} {{- /* ---------- asset publishing ---------- Every package asset is published under an npm-style path, /npm/@//, the same layout jsDelivr serves. Run `hugo mod npm pack` then `npm install` in the site root so node_modules/pseudocode is mounted at assets/node_modules/pseudocode/build." -}} {{- end -}} {{- /* Options follow the global options of pseudocode.js: https://github.com/SaswatPadhi/pseudocode.js#global-options Hugo lowercases param keys, so the dict cannot be jsonified directly: pseudocode.js expects camelCase names like indentSize. Rebuild the object with the exact keys it reads. A key the site omits is left out, so the library default applies. */ -}} {{- $pc := $tex.pseudocode | default dict -}} {{- if .Param "tex.pseudocode" -}} {{- $pc = merge $pc (.Param "tex.pseudocode") -}} {{- end -}} {{- $pcOpts := dict -}} {{- /* Keys are lowercase here because Hugo lowercases param names. */ -}} {{- with $pc.indentsize -}}{{- $pcOpts = merge $pcOpts (dict "indentSize" .) -}}{{- end -}} {{- with $pc.commentdelimiter -}}{{- $pcOpts = merge $pcOpts (dict "commentDelimiter" .) -}}{{- end -}} {{- with $pc.linenumberpunc -}}{{- $pcOpts = merge $pcOpts (dict "lineNumberPunc" .) -}}{{- end -}} {{- with $pc.titleprefix -}}{{- $pcOpts = merge $pcOpts (dict "titlePrefix" .) -}}{{- end -}} {{- if isset $pc "linenumber" -}}{{- $pcOpts = merge $pcOpts (dict "lineNumber" $pc.linenumber) -}}{{- end -}} {{- if isset $pc "noend" -}}{{- $pcOpts = merge $pcOpts (dict "noEnd" $pc.noend) -}}{{- end -}} {{- if isset $pc "scopelines" -}}{{- $pcOpts = merge $pcOpts (dict "scopeLines" $pc.scopelines) -}}{{- end -}} {{- if isset $pc "captioncount" -}}{{- $pcOpts = merge $pcOpts (dict "captionCount" $pc.captioncount) -}}{{- end -}} {{- /* The upstream stylesheet starts with `@import url(...KaTeX/0.16.7...)`, which would pull a second, mismatched KaTeX build from a CDN and fight the one this theme ships. Drop the import and rely on the stylesheet the active engine already loaded. */ -}} {{- $pcCSS = replaceRE `@import url\([^)]*\);\s*` "" $pcCSS.Content | resources.FromString "node_modules/pseudocode/build/pseudocode.min.css" | resources.Copy (printf "npm/pseudocode@%s/build/pseudocode.min.css" $pseudocodeVersion) -}} {{- $pcJS = $pcJS | resources.Copy (printf "npm/pseudocode@%s/build/%s" $pseudocodeVersion (path.Base $pcJS.Name)) -}} {{- /* jsonify escapes "<" to \u003c, so a stray "" in an option cannot break out. */ -}} {{- end -}} {{- /* ---------- engine ---------- */ -}} {{- if and (eq $engine "mathjax") (or $math $algorithm) -}} {{- $jsRes := resources.Get "node_modules/mathjax/es5/tex-chtml-full.js" -}} {{- if not $jsRes -}} {{- errorf "Alice: MathJax assets not found. Run `hugo mod npm pack` then `npm install` in the site root so node_modules/mathjax is mounted at assets/node_modules/mathjax/es5." -}} {{- end -}} {{- $js := $jsRes | resources.Copy (printf "npm/mathjax@%s/es5/%s" $mathjaxVersion (path.Base $jsRes.Name)) -}} {{- /* MathJax resolves its CHTML fonts relative to the script, so keep them under output/chtml/fonts/woff-v2 beside it. */ -}} {{- range resources.Match "node_modules/mathjax/es5/output/chtml/fonts/woff-v2/*.woff" -}} {{- $_ := (. | resources.Copy (printf "npm/mathjax@%s/es5/output/chtml/fonts/woff-v2/%s" $mathjaxVersion (path.Base .Name))).RelPermalink -}} {{- end -}} {{- else if and (eq $engine "katex") (or $math $algorithm) -}} {{- if $math -}} {{- end -}} {{- else if $isHugo -}} {{- /* The build-time work happens in render-passthrough.html (delimited snippets) and render-codeblock-math.html (fenced blocks), so the page loads no client math engine.
