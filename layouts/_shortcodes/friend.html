{{- /* Friend shortcode — render one entry from data/friends. Usage: {{< friend id="alice" >}} The `id` must match an `id` in a data file under data/friends. The friend fields are mapped onto card.html with variant="friend", so the friend card skin and markup stay the same while the renderer is shared. */ -}} {{- $friendId := .Get "id" | default "" -}} {{- with index hugo.Data.friends $friendId -}} {{- $friend := . -}} {{- /* Avatar: url → , svg → raw inline SVG, text → Markdown. The image stays decorative because the card title names the friend. */ -}} {{- $avatarHTML := "" -}} {{- with $friend.avatar -}} {{- $kind := lower (printf "%v" (.kind | default "url")) -}} {{- if eq $kind "url" -}} {{- with .content -}} {{- $avatarHTML = printf `` (htmlEscape (printf "%v" .)) -}} {{- end -}} {{- else if eq $kind "svg" -}} {{- $avatarHTML = printf "%v" (.content | default "") -}} {{- else if eq $kind "text" -}} {{- $avatarHTML = .content | default "" | markdownify -}} {{- end -}} {{- end -}} {{- $class := "" -}} {{- if $friend.dead -}}{{- $class = "dead" -}}{{- end -}} {{- partial "card.html" (dict "variant" "friend" "title" ($friend.name | default "") "description" ($friend.tagline | default "") "coverHTML" $avatarHTML "link" ($friend.website | default "") "target" "_blank" "id" ($friend.id | default "") "class" $class ) -}} {{- end -}}
