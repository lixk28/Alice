{{- /* One image slide inside a {{< carousel >}}. Args (all optional): image — page resource, global resource, or URL alt — image alt text (falls back to the caption when omitted) caption — overlay title for the slide (inline Markdown) description — overlay description under the caption (inline Markdown) captionPosition — per-slide caption position: bottom-left|bottom-center|bottom-right| top-left|top-center|top-right (default: carousel value) link — make the image a link (overrides the page lightbox) target — link target rel — link rel (defaults to "noopener noreferrer" for _blank) width — image width attribute height — image height attribute loading — image loading: lazy|eager (first slide defaults to eager) contentMode — "scaleToFill" (default), "aspectFit", or "aspectFill" lightbox — open the image in the page lightbox: true|false class — extra classes on the slide A slide must live inside a {{< carousel >}}; it only contributes its image. */ -}} {{- $image := .Get "image" -}} {{- $caption := .Get "caption" -}} {{- $description := .Get "description" -}} {{- $captionPosition := "" -}} {{- with .Get "captionPosition" -}} {{- $captionPosition = partial "carousel/position.html" (dict "value" . "kind" "caption" "overlay" false ) -}} {{- end -}} {{- $src := partial "image-url.html" (dict "url" $image "page" $.Page) -}} {{- $alt := "" -}} {{- if and (reflect.IsMap .Params) (isset .Params "alt") -}} {{- $alt = .Get "alt" -}} {{- else -}} {{- with $caption -}}{{- $alt = . | markdownify | plainify -}}{{- end -}} {{- end -}} {{- $target := .Get "target" -}} {{- $rel := .Get "rel" -}} {{- if and (eq $target "_blank") (not $rel) -}}{{- $rel = "noopener noreferrer" -}}{{- end -}} {{- $lightbox := true -}} {{- with .Get "lightbox" -}} {{- if in (slice "false" "no" "0" "off") (lower (printf "%v" .)) -}}{{- $lightbox = false -}}{{- end -}} {{- end -}} {{- $data := dict "Src" $src "Alt" $alt "Caption" $caption "Description" $description "CaptionPosition" $captionPosition "Link" (.Get "link") "Target" $target "Rel" $rel "Width" (.Get "width") "Height" (.Get "height") "Loading" (.Get "loading") "ContentMode" (.Get "contentMode") "Lightbox" $lightbox "Class" (.Get "class") -}} {{- $parent := .Parent -}} {{- if and $parent (eq $parent.Name "carousel") -}} {{- if not ($parent.Scratch.Get "carouselSlides") -}} {{- $parent.Scratch.Set "carouselSlides" slice -}} {{- end -}} {{- $parent.Scratch.Add "carouselSlides" $data -}} {{- else -}} {{- warnf "slide shortcode: must be used inside a carousel" -}} {{- end -}}
